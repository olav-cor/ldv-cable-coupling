\section*{Purpose}
Aggregate the per-frequency results into a single \textbf{mean $\eta$ per dataset} over a chosen
band $[f_{\min},f_{\max}]$ and plot it against the \textbf{sag parameter $\Theta$}, overlaying the
theory $\eta=1/(1+\Theta)$. Marker colour = cable, shape = gap ($\lozenge\,5$, $\square\,10$, $\circ\,15\unit{cm}$),
open = sag variant.

\section*{Key equations}
\textbf{Band-mean efficiency} over $\mathcal F=\{f\in\texttt{TARGET\_FREQS}: f_{\min}\le f\le f_{\max}\}$:
\begin{equation}
\bar\eta=\frac{1}{|\mathcal F|}\sum_{f\in\mathcal F}\eta_{\mathrm{med}}(f),\qquad
\sigma_\eta=\operatorname{std}_{f\in\mathcal F}\,\eta_{\mathrm{med}}(f).
\end{equation}

\textbf{Sag parameter --- two sources} (\texttt{theta\_source}):
\begin{align}
\textbf{sag:}\quad &\Theta=\frac{A\,w_0^2}{8I}\ \xrightarrow{\text{circular}}\ \tfrac12(w_0/r)^2,\\
\textbf{material (Probst Eq.~14):}\quad &\Theta=\frac{\rho^2 g^2 A^3 L^8}{128\,\pi^8 E^2 I^3},
\quad\text{$E$ averaged over 3 tensile tests }(\pm\sigma_E).
\end{align}

\textbf{Theory curve:}
$\;\boxed{\eta_{\mathrm{pred}}=1/(1+\Theta)}$.

\textbf{Supporting beam relations} (\S\,diagnostics): predicted self-weight mid-span sag and the
fundamental flexural frequency
\begin{equation}
w_0^{\mathrm{stat}}=\frac{\rho A g L^4}{4\pi^4 E I},\qquad
\frac{w_0^{\mathrm{stat}}}{L}\ (\text{dimensionless}),\qquad
f_1=\frac{22.4}{2\pi L^2}\sqrt{\frac{EI}{\rho A}}.
\end{equation}

\section*{Derived outputs}
\begin{itemize}
\item Per dataset: $\Theta$ (sag \& material), $\eta_{\mathrm{pred}}$, $\bar\eta\pm\sigma_\eta$, $n_f$ used.
\item \textbf{\S4 figures:} $\bar\eta$ vs $\Theta$ (sag-based and material-based $\Theta$), theory overlay, error bars.
\item \textbf{\S5 table:} all datasets sorted by $\Theta$, colour-graded $\bar\eta$, with $\eta_{\mathrm{pred}}$ for comparison.
\item \textbf{\S6:} the same scatter across 4 bands $\times$ 2 strain methods $\times$ 2 estimators to see frequency dependence.
\item \textbf{Diagnostics:} sag comparison (vs.\ LDV resolution $\sim200\,\mu$m), $\Theta$ comparison, sag-vs-force-ratio plots.
\end{itemize}
