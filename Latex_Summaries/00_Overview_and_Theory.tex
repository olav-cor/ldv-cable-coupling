\section*{1\quad Experiment in one paragraph}
\begin{itemize}
\item \textbf{Goal:} quantify how vibrational energy couples from a shaker into cables of differing geometry, measured with a Laser Doppler Vibrometer (LDV).
\item \textbf{Excitation:} log-sweep $1$--$500\unit{Hz}$, $10\unit{s}$ sweep $+1\unit{s}$ buffer, shaker at one end.
\item \textbf{Design matrix:} 7 cables $\times$ gap $\{5,10,15\}\unit{cm}$ $\times$ \{taut, sag, forced-sag, tape\}.
\item \textbf{Sampling:} $f_s = 5\unit{kHz}$, $N=60{,}000$ samples/record; LDV scans $\sim$9--17 points along the gap, 3 velocity components $(v_x,v_y,v_z)$.
\item \textbf{Central quantity:} \emph{coupling / strain-transfer efficiency}
$\;\eta=\dfrac{\text{cable axial elongation}}{\text{imposed span change}}$, compared against the sag parameter $\Theta$.
\end{itemize}

\section*{3\quad Signal primitives (used by every notebook)}
\textbf{Log-sweep arrival time} --- time at which the sweep reaches instantaneous frequency $f$:
\begin{equation}
t_c(f)=t_{\mathrm{buf}}+T_{\mathrm{sweep}}\,\frac{\ln(f/f_0)}{\ln(f_1/f_0)},\qquad
t_{\mathrm{buf}}=1\unit{s},\;T_{\mathrm{sweep}}=10\unit{s},\;f_0=1,\;f_1=500\unit{Hz}.
\end{equation}

\textbf{Zero-phase bandpass} (Butterworth, order $4$, \texttt{sosfiltfilt}), edges
$\;[f_{\mathrm{lo}},f_{\mathrm{hi}}]=[\,f-5,\;f+5\,]\unit{Hz}$ (absolute mode) or $f(1\pm0.5)$ (fractional mode), $f_{\mathrm{lo}}\ge0.5\unit{Hz}$.

\textbf{Frequency-domain integration} --- velocity $\to$ displacement, with a low-frequency floor that suppresses DC drift:
\begin{equation}
u(t)=\mathcal{F}^{-1}\!\left[\frac{V(f)}{i\,2\pi f_{\mathrm{safe}}}\right],
\qquad f_{\mathrm{safe}}=\max(f,\,f_{\mathrm{HP}}),\quad f_{\mathrm{HP}}=0.5\unit{Hz}.
\end{equation}

\textbf{Single-sided amplitude spectrum:}
$\;A(f)=\tfrac{2}{N}\,\big|\mathcal{F}\{x\}(f)\big|$ (DC and Nyquist not doubled).

\section*{4\quad Geometry, strain \& efficiency (definitions reused everywhere)}
\textbf{Chord.} For endpoint sensors $P_1,P_2$ spanning the gap:
$\;\hat{e}=(P_2-P_1)/L,\;\; L=\lVert P_2-P_1\rVert.$ Axial projection $u_\parallel=\mathbf{u}\cdot\hat e$.

\textbf{Static sag.} Perpendicular distance of each point from the chord,
$d_i=\big\lVert(\mathbf p_i-P_1)-[(\mathbf p_i-P_1)\cdot\hat e]\,\hat e\big\rVert$, $w_0=\max_i d_i$
(options: full 3-D vs.\ $x$--$z$ plane; optional constrained parabola
$d(s)=a\,s(s-L)$, vertex sag $w_0=-aL^2/4$).

\textbf{Three strain estimators} (axial strain $\varepsilon(s,t)$ along the gap):
\begin{itemize}
\item[\textbf{M1}] \emph{Spatial gradient} --- non-uniform central difference of $u_\parallel$;
$\;\delta x_\ell^{(1)}=\int_{\mathrm{gap}}\varepsilon\,\mathrm{d}s.$
\item[\textbf{M2}] \emph{3-D arc-length} --- per segment $\delta d_i=\hat e_i\cdot(\mathbf u_{i+1}-\mathbf u_i)$,
$\;\varepsilon_i=\delta d_i/L_{0,i}$, total $\delta x_\ell=\sum_i\delta d_i.$
\item[\textbf{M3}] \emph{Fourier gradient} --- on a uniform grid $\partial_s u=\mathcal F^{-1}\{ik\,\mathcal F\{u\}\}$,
$\;\delta x_\ell^{(3)}=\int_{\mathrm{gap}}\varepsilon\,\mathrm{d}s.$
\end{itemize}

\textbf{Coupling efficiency} (per frequency, against a reference span change $\delta_{\mathrm{ref}}$):
\begin{equation}
\eta(t)=\frac{\delta x_\ell(t)}{\delta_{\mathrm{ref}}(t)}\;\;\text{(median over window)},
\qquad
\eta_{\mathrm{env}}=\operatorname{med}\frac{|\mathcal H\{\delta x_\ell\}|}{|\mathcal H\{\delta_{\mathrm{ref}}\}|},
\end{equation}
where $\mathcal H$ is the Hilbert analytic envelope (phase-insensitive estimator). A floor at $5\%$ of $\max|\delta_{\mathrm{ref}}|$ blanks near-zero denominators. Reference options:
\begin{itemize}
\item \textbf{shaker:} $\delta L=\mathrm{proj}_{\hat e}(\mathbf u_{\mathrm{shaker}})$ (sign-flipped if shaker is the right end).
\item \textbf{ends:} $\delta_{\mathrm{ends}}=u_\parallel(\text{right})-u_\parallel(\text{left})$ --- cable-native span change.
\item \textbf{shaker\_end:} elongation of the sub-gap between shaker and the adjacent end sensor.
\end{itemize}

\section*{5\quad Theory model ($\Theta$, $\eta$, resonance) --- Probst 2026 / Simone draft}
\textbf{Cross-section.} Circular: $A=\pi r^2,\,I=\pi r^4/4$. Rectangular ribbon: $A=wh,\,I=wh^3/12$ (weak axis).

\textbf{Sag parameter $\Theta$ --- two routes:}
\begin{align}
\text{from measured sag:}\quad &\Theta=\frac{A\,w_0^2}{8I}\;\xrightarrow{\text{circular}}\;\tfrac12\!\left(\frac{w_0}{r}\right)^2,\\
\text{from material (Probst Eq.~14):}\quad &\Theta=\frac{\rho^2 g^2 A^3 L^8}{128\,\pi^8 E^2 I^3}.
\end{align}

\textbf{Quasi-static coupling efficiency} (flat below resonance):
\begin{equation}
\boxed{\;\eta_{\mathrm{pred}}=\frac{1}{1+\Theta}\;}
\end{equation}

\textbf{Fundamental flexural resonance} (clamped--clamped beam, Simone Eq.~18):
\begin{equation}
\omega_1=\frac{22.4}{L^2}\sqrt{\frac{EI}{\rho A}},\qquad f_1=\frac{\omega_1}{2\pi},\qquad 22.4=(4.730)^2.
\end{equation}
Above $f_1$ the model predicts constructive bending coupling that amplifies the transfer above $100\%$ (spike severity grows with $\Theta$).
