\section*{Purpose}
For every target frequency $f$, isolate the sweep around its arrival time, recover cable
\emph{displacement}, estimate axial \emph{strain} by three methods, and form the
\textbf{coupling efficiency} $\eta(f)=\delta x_\ell/\delta_{\mathrm{ref}}$. Output: one $\eta$-vs-$f$
curve per dataset vs.\ the quasi-static prediction $\eta=1/(1+\Theta)$.

\section*{Key equations}
\textbf{Window} centred on the sweep arrival time, $\pm n_{\mathrm{cyc}}$ cycles ($n_{\mathrm{cyc}}=5$):
\begin{equation}
t\in\big[\,t_c(f)-\tfrac{n_{\mathrm{cyc}}}{f},\;t_c(f)+\tfrac{n_{\mathrm{cyc}}}{f}\,\big],
\qquad t_c(f)=1+10\,\frac{\ln f}{\ln 500}\ \,[\mathrm{s}].
\end{equation}

\textbf{Integration} (per component, both cable and shaker):
$\;u=\mathcal F^{-1}\!\big[V/(i2\pi f_{\mathrm{safe}})\big],\ f_{\mathrm{safe}}=\max(f,0.5\unit{Hz}).$

\textbf{Reference span change} from shaker, projected on the chord:
\begin{equation}
\delta L(t)=\big(\mathbf u_{\mathrm{sh}}\cdot\hat e\big)\times(-1\ \text{if shaker on right}),
\qquad \varepsilon_{\mathrm{ref}}=\delta L/L_{\mathrm{chord}}.
\end{equation}

\textbf{Strain --- M1 (non-uniform central difference)} on axial $u_\parallel$, $\Delta s_+\!=\!s_{i+1}\!-\!s_i,\ \Delta s_-\!=\!s_i\!-\!s_{i-1}$:
\begin{equation}
\varepsilon_i=\frac{\Delta s_-}{\Delta s_+(\Delta s_+\!+\!\Delta s_-)}u_{i+1}
+\frac{\Delta s_+\!-\!\Delta s_-}{\Delta s_+\Delta s_-}u_{i}
-\frac{\Delta s_+}{\Delta s_-(\Delta s_+\!+\!\Delta s_-)}u_{i-1},
\quad \delta x_\ell^{(1)}=\!\!\int_{\mathrm{gap}}\!\!\varepsilon\,\mathrm{d}s.
\end{equation}

\textbf{Strain --- M2 (3-D arc-length, recommended)}:
$\;\delta d_i=\hat e_i\!\cdot(\mathbf u_{i+1}-\mathbf u_i),\quad \varepsilon_i=\delta d_i/L_{0,i},\quad \delta x_\ell=\sum_i\delta d_i.$

\textbf{Strain --- M3 (Fourier gradient)}: resample to uniform grid, $\partial_s u=\mathcal F^{-1}\{ik\,\mathcal F\{u\}\}$, integrate over gap.

\textbf{Efficiency} (per reference, median over window; plus phase-insensitive envelope variant):
\begin{equation}
\eta(t)=\frac{\delta x_\ell(t)}{\delta_{\mathrm{ref}}(t)},\qquad
\eta_{\mathrm{med}}=\operatorname{med}_t\,\eta(t),\qquad
\eta_{\mathrm{env}}=\operatorname{med}_t\frac{|\mathcal H\{\delta x_\ell\}|}{|\mathcal H\{\delta_{\mathrm{ref}}\}|}.
\end{equation}
