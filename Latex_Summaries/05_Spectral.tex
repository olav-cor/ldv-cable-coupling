\section*{Key equations}
\textbf{Single-sided amplitude spectrum} ($N$ samples, DC \& Nyquist not doubled):
\begin{equation}
A(f_k)=\frac{2}{N}\,\big|\mathcal F\{x\}(f_k)\big|,\qquad f_k=\frac{k f_s}{N}.
\end{equation}

\textbf{Dominant peaks} --- on the sensor-mean spectrum, normalised to the band max; peaks via
\texttt{scipy.find\_peaks} with min.\ height $0.02$ and min.\ separation $5\unit{Hz}$, top $n$ returned.

\textbf{Spectrogram (STFT magnitude)} --- Hann window, $n_{\mathrm{perseg}}=f_s/2$ ($\approx2\unit{Hz}$
resolution), $90\%$ overlap:
\begin{equation}
S(f,\tau)=\Big|\sum_{n} x[n]\,w[n-\tau]\,e^{-i2\pi f n/f_s}\Big|,\qquad
S_{\mathrm{dB}}=10\log_{10}\!\frac{S^2}{\max S^2}.
\end{equation}

\textbf{(Available primitive) F--K spectrum} --- 2-D FFT over $(t,x)$ after interpolation to a uniform
spatial grid and 2-D Hann taper: $\mathrm{FK}(f,k)$ in dB (frequency--wavenumber dispersion).

\section*{Options \& outputs}
\begin{itemize}
\item \textbf{Amplitude scaling}: \texttt{share\_scale=}\texttt{dataset} (one $y$-limit per dataset) or \texttt{component} (per axis).
\item \textbf{Spectra}: linear/log axes, \texttt{overlay\_sensors} (all sensors faint + bold mean), $f_{\max}$.
\item \textbf{Peaks}: \texttt{n\_peaks}, $f_{\max}$; separate analysis for cable channels ($v_{x,y,z}$) and shaker channels ($sh\_v_{x,y,z}$).
\item \textbf{Spectrograms}: \texttt{nperseg}, $f_{\max}$, \texttt{db\_floor}; per-sensor selection; interactive browser.
\item \textbf{Bare-shaker reference} (\texttt{DIRECT\_SHAKER\_REF}): same $1$--$500\unit{Hz}$ sweep with no cable, for normalising cable-loaded measurements; the sweep appears as a curved ridge $\sim1\!\to\!500\unit{Hz}$ over $10\unit{s}$.
\item \textbf{Widgets}: dataset/sensor/component time-vs-spectrum browser; spectrogram browser (mixes cable + shaker sources).
\end{itemize}
